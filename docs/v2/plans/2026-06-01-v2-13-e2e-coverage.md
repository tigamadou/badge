## Conventions générales (toutes tasks)

### Imports

```ts
import { test, expect } from "@playwright/test";
import { ... } from "../helpers/db";       // pour les specs dans e2e/<espace>/
import { loginAs, loginAsStaff } from "../helpers/auth";
import { mailpit } from "../helpers/mailpit";
import { expectPdfResponse } from "../helpers/pdf";
import { expectLastActivity } from "../helpers/audit";
```

### Pattern de spec

```ts
test.describe("<Domaine>", () => {
  test.afterAll(async () => {
    await cleanupE2ETestData();
    await disconnectDb();
  });

  test("<parcours golden>", async ({ page }) => {
    // 1. Setup Prisma
    const owner = await createTestStaff({ role: "STAFF_ADMIN", scope: "..." });
    // 2. Login (UI)
    await loginAs(page, owner);
    // 3. Mailpit reset si on attend un mail
    await mailpit.clear();
    // 4. Parcours UI
    await page.goto("/staff/organizations/new");
    // ...
    // 5. Assertions UI + BDD + Mailpit + audit
    await expect(page.getByText(/créée/i)).toBeVisible();
    await mailpit.expectMail({ to: ownerEmail, subjectPattern: /Bienvenue/ });
    await expectLastActivity({ type: "ORG_CREATED", userId: owner.id });
  });
});
```

### Sélecteurs

Préférer dans cet ordre : `getByRole(..., { name })`, `getByLabel(...)`, `getByTestId(...)`. Pas de sélecteurs CSS fragiles.

### Cleanup

Toujours en `afterAll`, jamais en `afterEach`, pour ne pas perdre l'état entre `test()` qui s'enchaînent dans un même describe.

## Annexe — Convention d'écriture des `test()` négatifs RBAC

Pattern à appliquer dans toutes les `rbac.spec.ts` :

```ts
test("OPERATOR ne peut pas archiver event", async ({ page, request }) => {
  const { eventId, orgId } = await createTestOrgWithEvent({ scope: "rbac-archive" });
  const op = await createTestEventMember({ scope: "rbac-archive-op", eventId, orgId, role: "OPERATOR" });
  await loginAs(page, op);

  // UI : bouton absent
  await page.goto(`/admin/${orgId}/events/${eventId}/settings`);
  await expect(page.getByRole("button", { name: /archiver/i })).toHaveCount(0);

  // API : 403
  const res = await page.request.post(`/api/admin/${orgId}/events/${eventId}/archive`);
  expect(res.status()).toBe(403);

  // Audit : aucun log d'archive créé
  await expectNoActivity({ type: "EVENT_ARCHIVED", eventId });
});
```
